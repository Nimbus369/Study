"""使用 AlexNet 训练 Fashion-MNIST。

运行示例：
    python temp.txt --epochs 10 --batch-size 128

Fashion-MNIST 原始图片是 28x28 的单通道灰度图。为了使用经典 AlexNet，
这里先把图片缩放到 224x224；网络第一层仍然使用 1 个输入通道。
"""

from __future__ import annotations

import argparse
import random
from pathlib import Path

import torch
from torch import nn
from torch.utils.data import DataLoader
from torchvision import datasets, transforms


NUM_CLASSES = 10
IMAGE_SIZE = 224


def set_seed(seed: int) -> None:
    """固定随机种子，方便复现实验结果。"""
    random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


def build_alexnet(num_classes: int = NUM_CLASSES) -> nn.Sequential:
    """构造用于 Fashion-MNIST 的经典 AlexNet。"""
    return nn.Sequential(
        # 使用较大的窗口和步幅捕捉大范围特征，同时降低空间尺寸。
        nn.Conv2d(1, 96, kernel_size=11, stride=4, padding=1),
        nn.ReLU(),
        nn.MaxPool2d(kernel_size=3, stride=2),

        # 使用 padding=2 保持卷积前后的高度和宽度不变。
        nn.Conv2d(96, 256, kernel_size=5, padding=2),
        nn.ReLU(),
        nn.MaxPool2d(kernel_size=3, stride=2),

        # 三个连续的 3x3 卷积层。
        nn.Conv2d(256, 384, kernel_size=3, padding=1),
        nn.ReLU(),
        nn.Conv2d(384, 384, kernel_size=3, padding=1),
        nn.ReLU(),
        nn.Conv2d(384, 256, kernel_size=3, padding=1),
        nn.ReLU(),
        nn.MaxPool2d(kernel_size=3, stride=2),

        nn.Flatten(),
        # 对 224x224 输入，最后的特征图大小为 256x5x5=6400。
        nn.Linear(256 * 5 * 5, 4096),
        nn.ReLU(),
        nn.Dropout(p=0.5),
        nn.Linear(4096, 4096),
        nn.ReLU(),
        nn.Dropout(p=0.5),
        nn.Linear(4096, num_classes),
    )


def print_model_shapes(model: nn.Module, device: torch.device) -> None:
    """用一个样例检查各层输出形状，便于发现尺寸不匹配。"""
    x = torch.randn(1, 1, IMAGE_SIZE, IMAGE_SIZE, device=device)
    print("各层输出形状：")
    with torch.no_grad():
        for layer in model:
            x = layer(x)
            print(f"{layer.__class__.__name__:>12}: {tuple(x.shape)}")


def train_one_epoch(
    model: nn.Module,
    data_loader: DataLoader,
    loss_function: nn.Module,
    optimizer: torch.optim.Optimizer,
    device: torch.device,
) -> tuple[float, float]:
    model.train()
    total_loss = 0.0
    total_samples = 0
    correct = 0

    for images, labels in data_loader:
        images = images.to(device, non_blocking=True)
        labels = labels.to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)
        outputs = model(images)
        loss = loss_function(outputs, labels)
        loss.backward()
        optimizer.step()

        batch_size = images.size(0)
        total_loss += loss.item() * batch_size
        total_samples += batch_size
        correct += (outputs.argmax(dim=1) == labels).sum().item()

    return total_loss / total_samples, 100.0 * correct / total_samples


@torch.inference_mode()
def evaluate(
    model: nn.Module,
    data_loader: DataLoader,
    loss_function: nn.Module,
    device: torch.device,
) -> tuple[float, float]:
    model.eval()
    total_loss = 0.0
    total_samples = 0
    correct = 0

    for images, labels in data_loader:
        images = images.to(device, non_blocking=True)
        labels = labels.to(device, non_blocking=True)
        outputs = model(images)
        loss = loss_function(outputs, labels)

        batch_size = images.size(0)
        total_loss += loss.item() * batch_size
        total_samples += batch_size
        correct += (outputs.argmax(dim=1) == labels).sum().item()

    return total_loss / total_samples, 100.0 * correct / total_samples


def make_data_loaders(
    data_root: Path,
    batch_size: int,
    num_workers: int,
    pin_memory: bool,
) -> tuple[DataLoader, DataLoader]:
    """下载（若尚未存在）并创建 Fashion-MNIST 数据加载器。"""
    transform = transforms.Compose(
        [
            transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)),
            transforms.ToTensor(),
            # Fashion-MNIST 的近似均值和标准差。
            transforms.Normalize((0.2860,), (0.3530,)),
        ]
    )

    train_dataset = datasets.FashionMNIST(
        root=str(data_root),
        train=True,
        download=True,
        transform=transform,
    )
    test_dataset = datasets.FashionMNIST(
        root=str(data_root),
        train=False,
        download=True,
        transform=transform,
    )

    loader_kwargs = {
        "batch_size": batch_size,
        "num_workers": num_workers,
        "pin_memory": pin_memory,
    }
    train_loader = DataLoader(train_dataset, shuffle=True, **loader_kwargs)
    test_loader = DataLoader(test_dataset, shuffle=False, **loader_kwargs)
    return train_loader, test_loader


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="训练 Fashion-MNIST AlexNet")
    parser.add_argument("--epochs", type=int, default=10, help="训练轮数，默认 10")
    parser.add_argument("--batch-size", type=int, default=128, help="批大小，默认 128")
    parser.add_argument("--lr", type=float, default=0.01, help="学习率，默认 0.01")
    parser.add_argument("--num-workers", type=int, default=0, help="数据加载进程数，Windows 建议 0")
    parser.add_argument("--seed", type=int, default=42, help="随机种子")
    parser.add_argument(
        "--data-root",
        type=Path,
        default=Path(__file__).resolve().parent / "data",
        help="数据集目录",
    )
    parser.add_argument(
        "--checkpoint",
        type=Path,
        default=Path(__file__).resolve().parent / "best_alexnet_from_scratch_fashionmnist.pth",
        help="最佳模型保存路径",
    )
    parser.add_argument(
        "--no-shape-check",
        action="store_true",
        help="不打印各层输出形状",
    )
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    if args.epochs < 1:
        raise ValueError("--epochs 必须是大于等于 1 的整数")
    if args.batch_size < 1:
        raise ValueError("--batch-size 必须是正整数")

    set_seed(args.seed)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"使用设备：{device}")
    if device.type == "cuda":
        print(f"GPU：{torch.cuda.get_device_name(0)}")

    train_loader, test_loader = make_data_loaders(
        data_root=args.data_root,
        batch_size=args.batch_size,
        num_workers=args.num_workers,
        pin_memory=device.type == "cuda",
    )

    model = build_alexnet().to(device)
    if not args.no_shape_check:
        print_model_shapes(model, device)

    loss_function = nn.CrossEntropyLoss()
    optimizer = torch.optim.SGD(
        model.parameters(),
        lr=args.lr,
        momentum=0.9,
        weight_decay=5e-4,
    )
    scheduler = torch.optim.lr_scheduler.StepLR(
        optimizer,
        step_size=5,
        gamma=0.1,
    )

    best_accuracy = 0.0
    args.checkpoint.parent.mkdir(parents=True, exist_ok=True)

    for epoch in range(args.epochs):
        train_loss, train_accuracy = train_one_epoch(
            model, train_loader, loss_function, optimizer, device
        )
        test_loss, test_accuracy = evaluate(
            model, test_loader, loss_function, device
        )
        scheduler.step()

        print(
            f"Epoch {epoch + 1:02d}/{args.epochs} | "
            f"lr={optimizer.param_groups[0]['lr']:.5f} | "
            f"训练损失={train_loss:.4f} | 训练准确率={train_accuracy:.2f}% | "
            f"测试损失={test_loss:.4f} | 测试准确率={test_accuracy:.2f}%"
        )

        if test_accuracy > best_accuracy:
            best_accuracy = test_accuracy
            torch.save(model.state_dict(), args.checkpoint)
            print(f"已保存最佳模型：{args.checkpoint}（{best_accuracy:.2f}%）")

    print(f"训练完成，最佳测试准确率：{best_accuracy:.2f}%")


if __name__ == "__main__":
    main()
